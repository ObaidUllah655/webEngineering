Mid Task 2

JSON Data Handling

JSON.stringify() - JSON.parse() - Object Display

Step 1 — Create a JavaScript Object

// Create a JavaScript object with name, age, course
const student = {
  name:   "Obaid Ullah",
  age:    21,
  course: "Web Engineering"
};

console.log("Original Object:", student);
console.log("  name  :", student.name);
console.log("  age   :", student.age);
console.log("  course:", student.course);
Output

Step 2 — Convert Object to JSON (JSON.stringify)

// JSON.stringify() converts a JavaScript object to a JSON string
const jsonString = JSON.stringify(student);
console.log("JSON String:", jsonString);

// Pretty-print with indentation
const prettyJSON = JSON.stringify(student, null, 2);
console.log("Pretty JSON:\n" + prettyJSON);
Output

Step 3 — Convert JSON back to Object (JSON.parse)

// JSON.parse() converts a JSON string back to a JavaScript object
const parsedObject = JSON.parse(jsonString);
console.log("Parsed Object:", parsedObject);
console.log("Type:", typeof parsedObject);   // object
console.log("name :", parsedObject.name);
console.log("age  :", parsedObject.age);
console.log("course:", parsedObject.course);
Output
