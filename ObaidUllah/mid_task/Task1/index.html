ES6 Features and Classes

var vs let vs const · Arrow Functions · ES6 Class

Section 1 — var, let, and const

// var: function-scoped, can be re-declared
var x = 10;
var x = 20; // no error
console.log("var x:", x); // 20

// let: block-scoped, cannot be re-declared
let age = 21;
age = 22; // re-assignment allowed
// let age = 25; // ERROR: already declared
console.log("let age:", age); // 22

// const: block-scoped, cannot be re-assigned
const PI = 3.14159;
// PI = 3; // ERROR: Assignment to constant variable
console.log("const PI:", PI); // 3.14159

Section 2 — Arrow Function: Square of a Number

// Arrow function that calculates the square of a number
const square = (num) => num * num;

console.log("square(4)  =", square(4));   // 16
console.log("square(7)  =", square(7));   // 49
console.log("square(12) =", square(12));  // 144

Section 3 — Class: Student

class Student {
  constructor(name, marks) {
    this.name  = name;
    this.marks = marks;
  }

  display() {
    const avg = (this.marks.reduce((a, b) => a + b, 0) / this.marks.length).toFixed(1);
    return `Name: ${this.name} | Marks: [${this.marks.join(", ")}] | Average: ${avg}`;
  }
}

const s1 = new Student("Obaid", [85, 90, 78, 92]);
const s2 = new Student("Ahmed", [70, 65, 80, 75]);

console.log(s1.display());
console.log(s2.display());
